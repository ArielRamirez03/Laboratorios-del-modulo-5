# PRACTICA 3 - SERVIDOR 1
# Usuario: arielrc
# IP: 10.0.0.36

# Verificar IP
ip -4 addr show ens160

# Instalar Apache y Keepalived
sudo dnf install httpd keepalived -y

# Iniciar Apache
sudo systemctl enable --now httpd
sudo systemctl status httpd

# Crear pagina HTML
echo '<h1>Servidor 1 - Principal</h1>' | sudo tee /var/www/html/index.html

# Configurar firewall
sudo firewall-cmd --permanent --add-service=http
sudo firewall-cmd --permanent --add-rich-rule='rule protocol value="vrrp" accept'
sudo firewall-cmd --reload

# Crear script de comprobacion
sudo tee /usr/local/bin/check_httpd.sh > /dev/null <<'EOF'
#!/bin/bash
/usr/bin/systemctl is-active --quiet httpd
EOF

sudo chmod 755 /usr/local/bin/check_httpd.sh

# Configurar Keepalived
sudo nano /etc/keepalived/keepalived.conf
global_defs {
    router_id SERVER1
    script_user root
    enable_script_security
}

vrrp_script chk_httpd {
    script "/usr/local/bin/check_httpd.sh"
    interval 2
    fall 2
    rise 2
}

vrrp_instance VI_1 {
    state BACKUP
    interface ens160
    virtual_router_id 51
    priority 150
    advert_int 1

    virtual_ipaddress {
        10.0.0.100/24 dev ens160
    }

    track_script {
        chk_httpd
    }
}
# Verificar configuracion
sudo keepalived -t

# Iniciar Keepalived
sudo systemctl enable --now keepalived

# Verificar servicio
sudo systemctl status keepalived

# Verificar IP flotante
ip -4 addr show ens160

# Apagar servidor para probar HA
sudo shutdown -h now
